# Password field

Text field with a show/hide button and password-manager friendly attributes.

Group: Forms. Registry item: http://localhost:5173/r/password-field.json

## Install

```sh
bunx shadcn-vue@latest add http://localhost:5173/r/password-field.json
```

Installs with: theme, utils, field, announce. npm packages: none beyond Vue.

Import it from your own copy: `import Passwordfield from '@/components/ui/password-field.vue'`.

## Props

| Prop | Type | Default | What it does |
| --- | --- | --- | --- |
| `v-model` | `string` | `''` | Two-way bound value. |
| `label (required)` | `string` |  |  |
| `hint` | `string` |  |  |
| `error` | `string` |  |  |
| `required` | `boolean` |  |  |
| `optional` | `boolean` |  |  |
| `hideLabel` | `boolean` |  |  |
| `readonly` | `boolean` |  |  |
| `disabled` | `boolean` |  |  |
| `loading` | `boolean` |  |  |
| `mode` | `'current' \| 'new'` | `'current'` | 'current' for signing in, 'new' for choosing one: it sets autocomplete so a password manager does the right thing. |
| `rules` | `{ label: string; test: (value: string) => boolean }[]` |  | A checklist shown under the field: each rule is met when `test` returns true. |

## Usage

```vue
<script setup lang="ts">
import { computed, ref } from 'vue'
import Button from '@/components/ui/button.vue'
import Card from '@/components/ui/card.vue'
import PasswordField from '@/components/ui/password-field.vue'

const current = ref('')
const short = ref('abc')
const error = computed(() => (short.value && short.value.length < 8 ? 'Use at least 8 characters.' : undefined))
const fresh = ref('')
const rules = [
  { label: 'At least 8 characters', test: (v: string) => v.length >= 8 },
  { label: 'A number', test: (v: string) => /\d/.test(v) },
  { label: 'An upper-case letter', test: (v: string) => /[A-Z]/.test(v) },
]
</script>

<template>
  <div class="grid items-start gap-6 md:grid-cols-2">
    <PasswordField v-model="current" label="Password" placeholder="Reveal with the eye" />
    <PasswordField v-model="short" label="Error" :error="error" />
    <PasswordField label="Disabled" model-value="locked" disabled />
    <Card title="Choose a password">
      <PasswordField v-model="fresh" label="New password" mode="new" :rules="rules" />
      <div><Button :disabled="rules.some((r) => !r.test(fresh))">Save</Button></div>
    </Card>
  </div>
</template>
```
